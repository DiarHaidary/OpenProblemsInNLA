open Lean Elab Command in
run_cmd do
  let env ← getEnv
  let mut pending := [``NLA.RA20.criticalCountConjecture]
  let mut seen : List Name := []
  let mut allUsed : List Name := []
  for _ in [:1000] do
    match pending with
    | [] => pure ()
    | name :: rest =>
      pending := rest
      unless seen.contains name do
        seen := name :: seen
        let some ci := env.find? name | throwError "Missing definition {name}"
        if ci.isUnsafe || ci.isPartial then throwError "Unsafe/partial definition {name}"
        let some body := ci.value? (allowOpaque := true) | throwError "Bodyless definition {name}"
        let used := ci.type.getUsedConstants.toList ++ body.getUsedConstants.toList
        allUsed := used ++ allUsed
        for ax in (← liftCoreM <| collectAxioms name) do
          unless [``propext, ``Classical.choice, ``Quot.sound].contains ax do
            throwError "Forbidden definition axiom: {name}: {ax}"
        let next := used.filter (fun n => n.toString.startsWith "NLA.RA20.")
        logInfo m!"ACTUAL_DEFINITION_EDGE {name}: {next}"
        pending := next ++ pending
  unless pending.isEmpty do throwError "Incomplete definition traversal"
  for need in [``NLA.RA20.variety, ``NLA.RA20.definingIdeal,
      ``NLA.RA20.SmoothPoint, ``NLA.RA20.TangentVector,
      ``NLA.RA20.fullFrobeniusDistance, ``NLA.RA20.HasCriticalCount,
      ``NLA.RA20.HasGenericCriticalCount, ``Algebra.smoothLocus,
      ``MvPolynomial.vanishingIdeal, ``MvPolynomial.pointToPoint,
      ``MvPolynomial.pderiv, ``fderiv, ``Matrix.rank, ``Cardinal.mk] do
    unless allUsed.contains need do throwError "Missing actual semantic dependency {need}"
    logInfo m!"REQUIRED_DEFINITION: {need}"
  for forbidden in [``NLA.RA20.ExactlyOneZero, ``NLA.RA20.candidate,
      ``NLA.RA20.GenericData, ``NLA.RA20.abcIdeal, ``sorryAx] do
    if allUsed.contains forbidden then throwError "Witness predicate defines target: {forbidden}"
  logInfo m!"INSPECTED_TARGET_DEFINITIONS: {seen.length}"
