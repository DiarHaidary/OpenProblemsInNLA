import Mathlib.RingTheory.Smooth.Locus
import LeanCert.Tactic.Verification

set_option autoImplicit false
set_option leancert.trust "kernel"

namespace NLA.RA20

/-- An algebra isomorphism carries the actual local algebraic smooth locus to
the actual smooth locus. No smoothness or finite-presentation premise is added. -/
theorem smoothLocus_comap_algEquiv {R A B : Type*}
    [CommRing R] [CommRing A] [CommRing B] [Algebra R A] [Algebra R B]
    (e : A ≃ₐ[R] B) (p : PrimeSpectrum B) :
    PrimeSpectrum.comap e.toRingHom p ∈ Algebra.smoothLocus R A ↔
      p ∈ Algebra.smoothLocus R B := by
  let q := PrimeSpectrum.comap e.toRingHom p
  have hmap : Submonoid.map e q.asIdeal.primeCompl = p.asIdeal.primeCompl := by
    ext b
    constructor
    · rintro ⟨a, ha, rfl⟩
      exact ha
    · intro hb
      refine ⟨e.symm b, ?_, e.apply_symm_apply b⟩
      change e (e.symm b) ∉ p.asIdeal
      simpa only [e.apply_symm_apply] using hb
  change Algebra.FormallySmooth R (Localization.AtPrime q.asIdeal) ↔
    Algebra.FormallySmooth R (Localization.AtPrime p.asIdeal)
  exact Algebra.FormallySmooth.iff_of_equiv
    (IsLocalization.algEquivOfAlgEquiv
      (Localization.AtPrime q.asIdeal) (Localization.AtPrime p.asIdeal) e hmap)

#assert_trust kernel smoothLocus_comap_algEquiv
#print axioms smoothLocus_comap_algEquiv

end NLA.RA20
